public void numeros(int[] arr) {
    for (int i = 0; i < arr.length; i++) {
        System.out.print(arr[i]);
        if (arr[i] > 0) System.out.print(" ");
        for (int j = 1; j <= arr[i]; j++) {
            System.out.print("#");
            if (j < arr[i]) System.out.print("-");
        }
        System.out.println();
    }
}
